cmake_minimum_required(VERSION 3.16)
project(KalynaLab LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(kalyna STATIC Kalyna-reference/kalyna.c Kalyna-reference/tables.c)
target_include_directories(kalyna PUBLIC Kalyna-reference)
file(GLOB SPEED_KALYNA CONFIGURE_DEPENDS ciphers-speed/Kalyna/*.cpp)
add_executable(kalyna-lab lab/main.cpp lab/compare.cpp ${SPEED_KALYNA} ciphers-speed/aes/aes.cpp ciphers-speed/time_measure.cpp)
target_compile_definitions(kalyna-lab PRIVATE KALYNA_EMBED_COMPARE)
target_link_libraries(kalyna-lab PRIVATE kalyna)
enable_testing()
add_test(NAME lab-known-answers-and-tampering COMMAND kalyna-lab test)
